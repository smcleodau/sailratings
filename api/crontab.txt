# Sailing Rating Data Platform — Cron Schedule
# Install with: crontab /home/irc-data/code/sailratings/api/crontab.txt
#
# All commands use the project venv. Logs rotate via find cleanup.
# MAILTO can be set for error notifications.

SHELL=/bin/bash
PATH=/home/irc-data/code/sailratings/api/.venv/bin:/usr/local/bin:/usr/bin:/bin
IRC_DATA_DIR=/home/irc-data/code/sailratings/api
LOG_DIR=/home/irc-data/logs

# === ORC (URGENT: never miss a day) ===
# Daily full ORC snapshot at 03:00 UTC
0 3 * * *    cd $IRC_DATA_DIR && irc-data scrape orc >> $LOG_DIR/orc_$(date +\%F).log 2>&1

# Refresh materialized views after ORC ingestion
0 4 * * *    cd $IRC_DATA_DIR && irc-data refresh-views >> $LOG_DIR/views_$(date +\%F).log 2>&1

# === IRC ===
# Daily TCC listing CSV download at 06:00 UTC
0 6 * * *    cd $IRC_DATA_DIR && irc-data scrape tcc >> $LOG_DIR/tcc_$(date +\%F).log 2>&1

# Weekly cert discovery (Sunday 07:00 UTC)
0 7 * * 0    cd $IRC_DATA_DIR && irc-data scrape certs --exhaustive >> $LOG_DIR/certs_$(date +\%F).log 2>&1

# Weekly PDF parsing (Sunday 08:00 UTC)
0 8 * * 0    cd $IRC_DATA_DIR && irc-data parse-certs >> $LOG_DIR/parse_$(date +\%F).log 2>&1

# === Race Results — All 13 SailSys clubs, every 30 min (incremental) ===
*/30 * * * *  cd $IRC_DATA_DIR && irc-data scrape results --source sailsys --all-clubs >> $LOG_DIR/sailsys_$(date +\%F_\%H\%M).log 2>&1

# Re-match unmatched results after each scrape cycle (offset by 20 min)
20,50 * * * *  cd $IRC_DATA_DIR && irc-data rematch-results >> $LOG_DIR/rematch_$(date +\%F_\%H\%M).log 2>&1

# --- UK / Ireland / Asia offshore results ---
# SailRaceHQ — current UK offshore results platform (replaces RORC static
# HTML which only covered 2007-2022). Weekly Tuesday 10:00.
0 10 * * 2   cd $IRC_DATA_DIR && irc-data scrape results --source sailracehq >> $LOG_DIR/sailracehq_$(date +\%F).log 2>&1

# ISORA — Irish Sea Offshore Racing Association. Weekly Tuesday 11:00.
0 11 * * 2   cd $IRC_DATA_DIR && irc-data scrape results --source isora >> $LOG_DIR/isora_$(date +\%F).log 2>&1

# RHKYC — Royal Hong Kong Yacht Club fleet. Weekly Wednesday 10:00 (timezone
# offset: Wednesday morning UTC is a sensible quiet time for the Asia fleet).
0 10 * * 3   cd $IRC_DATA_DIR && irc-data scrape results --source rhkyc >> $LOG_DIR/rhkyc_$(date +\%F).log 2>&1

# --- Decommissioned / on-demand sources (intentionally NOT scheduled) ---
# rorc:        static HTML only covers 2007-2022. Use sailracehq for post-2023.
# cowesweek:   annual event (early August). Run by hand during/after the event:
#                irc-data scrape results --source cowesweek --year YYYY
# sydneyhobart: annual event (26 December). Same pattern as Cowes Week.
# sailwave:    used by scattered clubs, no canonical URL pattern. Run by hand
#              against specific event URLs when needed.

# === Identity & Design Maintenance ===
# Weekly: match new ORC certs to IRC boats (Sunday 05:00 UTC)
0 5 * * 0    cd $IRC_DATA_DIR && irc-data match-boats >> $LOG_DIR/match_$(date +\%F).log 2>&1

# Monthly: re-seed designs and backfill (1st of month 04:30 UTC)
30 4 1 * *   cd $IRC_DATA_DIR && irc-data seed-designs >> $LOG_DIR/designs_$(date +\%F).log 2>&1

# === Historical backfill ===
# Monthly: Wayback Machine archive search (1st of month)
0 4 1 * *    cd $IRC_DATA_DIR && irc-data scrape wayback >> $LOG_DIR/wayback_$(date +\%F).log 2>&1

# === Monitoring ===
# Daily health check at 09:00 UTC with webhook notification
0 9 * * *    cd $IRC_DATA_DIR && irc-data health-check --notify >> $LOG_DIR/health_$(date +\%F).log 2>&1

# === Cleanup ===
# Delete logs older than 30 days (midnight)
0 0 * * *    find /home/irc-data/logs -name "*.log" -mtime +30 -delete
0 1 * * * /home/irc-data/scripts/sync-prod-db.sh >> /home/irc-data/logs/sync-prod-db.log 2>&1
@reboot nohup /home/irc-data/scripts/scrape-everything.sh >> /home/irc-data/logs/scrape-master.log 2>&1 &
@reboot nohup /home/irc-data/scripts/data-maintenance.sh >> /home/irc-data/logs/data-maintenance.log 2>&1 &
0 10 * * 1 source /home/irc-data/.env && python3 -m pytest tests/e2e_checkout_flow.py -v --timeout=120 >> /home/irc-data/logs/e2e_$(date +\%Y\%m\%d).log 2>&1
30 4 * * *   cd $IRC_DATA_DIR && irc-data seed-design-designers >> $LOG_DIR/identity_$(date +\%F).log 2>&1 && irc-data backfill-boat-identity >> $LOG_DIR/identity_$(date +\%F).log 2>&1
