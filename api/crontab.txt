# Sailing Rating Data Platform — Cron Schedule
# Install with: crontab /home/irc-data/code/sailratings/api/crontab.txt
#
# All commands use the project venv. Logs rotate via find cleanup.
# MAILTO can be set for error notifications.

SHELL=/bin/bash
PATH=/home/irc-data/code/sailratings/api/.venv/bin:/home/irc-data/.local/bin:/usr/local/bin:/usr/bin:/bin
IRC_DATA_DIR=/home/irc-data/code/sailratings/api
LOG_DIR=/home/irc-data/logs
OP_ENV="source /home/irc-data/.credentials/op-service-account.env && op run --environment vzhxzxt7mgb4tolyepo5wqzcz4 --"

# ---------------------------------------------------------------------------
# OPS-02-05 — crontab migrated to Temporal schedules.
#
# No production ingestion runs on cron any more.  Every collection source
# (ORC, TCC, SailSys, TopYacht, ISORA, RHKYC, SailRaceHQ, cert discovery /
# parse, wayback, plus the raw-capture feeds and maintenance jobs) is driven
# by a Temporal schedule — one ``source-<slug>`` schedule per ``data_sources``
# register row, reconciled by the OPS-01-02 ScheduleRegistry /
# ScheduleSyncLoopWorkflow and executed by the SourceRunWorkflow.
#
# Each source migrated one at a time: its Temporal schedule was enabled, the
# cron line kept until two consecutive green Temporal runs landed in the
# ``source_runs`` ledger, then the cron line was deleted.  Watchdog,
# health-check, refresh-views and log cleanup moved last.
#
# Every source run writes a ``source_runs`` row (the run ledger); the admin
# Scrapers page reads per-source health from there.
#
# The entries remaining below are host maintenance only — they are not
# ``irc-data`` ingestion (db sync, reboot bring-up, e2e smoke).  The
# commented RORC line is retained as historical documentation.
# ---------------------------------------------------------------------------

# === Historical note ===
# Tuesday 10:00 — UK results (RORC)
# Note: RORC static HTML only covers 2007-2022; post-2023 needs SailRaceHQ
# (superseded by the SailRaceHQ Temporal schedule; retained for reference)
# 0 10 * * 2   cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data scrape results --source rorc" >> $LOG_DIR/rorc_$(date +\%F).log 2>&1

# === Identity & Design Maintenance ===
# Weekly: match new ORC certs to IRC boats (Sunday 05:00 UTC)
0 5 * * 0    cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data match-boats" >> $LOG_DIR/match_$(date +\%F).log 2>&1

# Monthly: re-seed designs and backfill (1st of month 04:30 UTC)
30 4 1 * *   cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data seed-designs" >> $LOG_DIR/designs_$(date +\%F).log 2>&1

# === Historical backfill ===
# Monthly: Wayback Machine archive search (1st of month)
0 4 1 * *    cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data scrape wayback" >> $LOG_DIR/wayback_$(date +\%F).log 2>&1

# === Monitoring ===
# Staleness watchdog (OPS-01-04 / OPS-02-03): every 15 min
*/15 * * * * cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data scrape-watchdog" >> $LOG_DIR/watchdog_$(date +\%F).log 2>&1

# Daily health check at 06:00 UTC
0 6 * * *    cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data scraper-health" >> $LOG_DIR/scraper_health_$(date +\%F).log 2>&1

# AD-01-15 — nightly admin_metrics: completeness facts for the data-health
# page.  Runs before the morning health-check so the page renders from the
# precomputed stream (no heavy queries on request).  05:30 UTC nightly.
30 5 * * *   cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data admin-metrics" >> $LOG_DIR/admin_metrics_$(date +\%F).log 2>&1

# Daily heartbeat + dead-man ping at 09:00 UTC
0 9 * * *    cd $IRC_DATA_DIR && bash -c "$OP_ENV irc-data health-check --notify" >> $LOG_DIR/health_$(date +\%F).log 2>&1

# === Host maintenance ===
# Delete logs older than 30 days (midnight)
0 0 * * *    find /home/irc-data/logs -name "*.log" -mtime +30 -delete

# Nightly production DB sync
0 1 * * * /home/irc-data/scripts/sync-prod-db.sh >> /home/irc-data/logs/sync-prod-db.log 2>&1

# Reboot bring-up helpers
@reboot nohup /home/irc-data/scripts/scrape-everything.sh >> /home/irc-data/logs/scrape-master.log 2>&1 &
@reboot nohup /home/irc-data/scripts/data-maintenance.sh >> /home/irc-data/logs/data-maintenance.log 2>&1 &

# Weekly e2e checkout-flow smoke test (not ingestion)
0 10 * * 1 cd $IRC_DATA_DIR && bash -c "export PYTHONPATH=src && $OP_ENV python3 -m pytest tests/e2e_checkout_flow.py -v --timeout=120" >> /home/irc-data/logs/e2e_$(date +\%Y\%m\%d).log 2>&1
